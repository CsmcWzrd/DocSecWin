cmake_minimum_required(VERSION 3.16)
project(DocSecWin VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
set(H third_party/httpdwin)
set(Z third_party/Zar)
add_executable(DocSecWin src/main.cpp src/database.cpp src/security.cpp src/peer_client.cpp src/pull.cpp
 ${H}/docsec_http.cpp ${H}/httpresponse.cpp ${H}/httprequest.cpp
 ${H}/httpheaders.cpp ${H}/cookie.cpp
 ${Z}/src/zar_header.cpp ${Z}/src/zar_store.cpp ${Z}/src/zar_utils.cpp ${Z}/src/zar_stream.cpp)
target_include_directories(DocSecWin PRIVATE src ${H} ${Z}/include)
target_link_libraries(DocSecWin PRIVATE OpenSSL::Crypto OpenSSL::SSL Threads::Threads)
if(WIN32)
 target_link_libraries(DocSecWin PRIVATE ws2_32)
 target_compile_definitions(DocSecWin PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
option(BUILD_TESTING "Enable optional integration test client" OFF)
include(CTest)
if(BUILD_TESTING)
 add_executable(zar-stream-test tests/zar_stream_test.cpp ${Z}/src/zar_stream.cpp ${Z}/src/zar_store.cpp ${Z}/src/zar_header.cpp ${Z}/src/zar_utils.cpp)
 target_include_directories(zar-stream-test PRIVATE ${Z}/include)
 target_link_libraries(zar-stream-test PRIVATE Threads::Threads)
 add_test(NAME zar-stream COMMAND zar-stream-test)
 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_FOUND)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:DocSecWin>)
 endif()
endif()
install(TARGETS DocSecWin RUNTIME DESTINATION bin)
